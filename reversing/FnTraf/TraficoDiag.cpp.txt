extern "C" int ChooseDiag(char* buf, int n) {
    // histograma de 200 llamadas a CCarCtrl::ChooseModel (solo lee estado y consume rand del EXE) + valor forzado 0xA4448C
    int ids[10]; int cuenta[10]; int arg[10]; int nd = 0, menos1 = 0;
    for (int i = 0; i < 200; i++) {
        int32 a = -7;
        const int32 m = CCarCtrl::ChooseModel(&a);
        if (m == -1) { menos1++; continue; }
        int k = 0; while (k < nd && ids[k] != m) k++;
        if (k == nd) { if (nd == 10) continue; ids[nd] = m; cuenta[nd] = 0; arg[nd] = a; nd++; }
        cuenta[k]++;
    }
    int o = _snprintf_s(buf, n, _TRUNCATE, "  CHOOSE: forzado(0xA4448C)=%d noMenos1=%d de 200; -1=%d |", *(volatile int*)0xA4448C, 200 - menos1, menos1);
    for (int k = 0; k < nd && o > 0 && o < n - 40; k++) o += _snprintf_s(buf + o, n - o, _TRUNCATE, " modelo %d (tipo %d) x%d", ids[k], arg[k], cuenta[k]);
    int o2 = o;
    if (o2 > 0 && o2 < n - 80) { o2 += _snprintf_s(buf + o2, n - o2, _TRUNCATE, " | apropiados:"); for (unsigned i = 0; i < 6 && o2 < n - 20; i++) o2 += _snprintf_s(buf + o2, n - o2, _TRUNCATE, " %d", (int)CPopulation::m_AppropriateLoadedCars.GetMember(i)); }
    return o2;
}
