float CameraRound4(float value) {
    double scaled,integral;
    float result;
    const int five=5,four=4;
    __asm {
        fld qword ptr ds:[00859EA8h]
        fild five
        mov eax,00822130h
        call eax
        fmul value
        fld value
        fcomp dword ptr ds:[00858B50h]
        fnstsw ax
        test ah,5
        jp camera_round_positive
        fsub dword ptr ds:[00858C80h]
        jmp camera_round_fraction
    camera_round_positive:
        fadd dword ptr ds:[00858C80h]
    camera_round_fraction:
        fmul dword ptr ds:[00858B1Ch]
        fstp scaled
    }
    std::modf(scaled,&integral);
    __asm {
        fld integral
        fld qword ptr ds:[00859EA8h]
        fild four
        mov eax,00822130h
        call eax
        fdivp st(1),st
        fstp result
    }
    return result;
}
float CameraPedHeading(CPed* ped) {
    if (!*reinterpret_cast<CMatrix**>(reinterpret_cast<uint8*>(ped)+0x14)) return *reinterpret_cast<float*>(reinterpret_cast<uint8*>(ped)+0x10);
    const float x=-ped->GetMatrix().GetForward().x,y=ped->GetMatrix().GetForward().y;
    float result;
    __asm {
        fld x
        fld y
        fpatan
        fstp result
    }
    return result;
}
void CameraRoundVector4(CVector& v) { v.x=CameraRound4(v.x);v.y=CameraRound4(v.y);v.z=CameraRound4(v.z); }
}
